FUNCTION function placementPlan( @ 1322
function placementPlan(ceiling,questionCount=24){
  const eligible=Array.from({length:ceiling+1},(_,g)=>g); // Levels 0 through registered grade, inclusive.
  const perLevel=Math.floor(questionCount/eligible.length);
  let remaining=questionCount%eligible.length;
  const ids=[];
  // Extra questions go to the highest eligible levels, never above the saved grade.
  for(const g of eligible){
    const requested=perLevel+(g>ceiling-remaining?1:0);
    const pool=byGrade[g].filter(q=>q.mode==='mcq'&&q.type!=='Listening');
    if(pool.length<requested)throw new Error('Insufficient placement questions for level '+g);
    ids.push(...shuffle(pool).slice(0,requested).map(q=>q.id));
  }
  return shuffle(ids);
}
